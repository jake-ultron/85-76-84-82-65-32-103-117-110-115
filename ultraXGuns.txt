local function New(class, props, kids)
    local o = Instance.new(class)
    local isGui = o:IsA("GuiObject")
    if isGui then
        o.BorderSizePixel = 0
        if class == "Frame" or class == "TextButton" or class == "ScrollingFrame" then o.BackgroundColor3 = C.WHITE end
    end
    if o:IsA("GuiButton") then o.AutoButtonColor = false end
    if class == "TextLabel" or class == "TextButton" or class == "TextBox" then
        o.FontFace = FF("Medium"); o.TextColor3 = C.TEXT
    end
    local parent
    if props then
        for k, v in pairs(props) do
            if k == "Parent" then parent = v else o[k] = v end
        end
    end
    if kids then for _, k in ipairs(kids) do k.Parent = o end end
    if parent then o.Parent = parent end
    return o
end
local function Corner(r) return New("UICorner", { CornerRadius = UDim.new(0, r) }) end
local function Pad(t, l, r, b)
    return New("UIPadding", { PaddingTop = UDim.new(0, t), PaddingLeft = UDim.new(0, l or t), PaddingRight = UDim.new(0, r or l or t), PaddingBottom = UDim.new(0, b or t) })
end
local function List(dir, pad, ha, va)
    return New("UIListLayout", {
        FillDirection = dir or Enum.FillDirection.Vertical, SortOrder = Enum.SortOrder.LayoutOrder,
        Padding = UDim.new(0, pad or 0),
        HorizontalAlignment = ha or Enum.HorizontalAlignment.Left, VerticalAlignment = va or Enum.VerticalAlignment.Top,
    })
end
local function Grad(c0, c1, rot)

end
local function tw(o, t, p, style, dir)

end
local function safe(f, ...)
    if type(f) ~= "function" then return end
    local ok, e = pcall(f, ...)
    if not ok then warn("[UltraXUI] " .. tostring(e)) end
end
local function textColorFor(c)
    local br = 0.299 * c.R + 0.587 * c.G + 0.114 * c.B
    if br > 0.62 then return Color3.fromRGB(20, 20, 20) end
    return C.WHITE
end

        function T:Dropdown(o)
            local obj = { Values = o.Values or {}, Value = o.Value, Multi = o.Multi == true, AllowNone = o.AllowNone, Callback = o.Callback, Opened = false, __type = "Dropdown" }
            if obj.Multi then
                if type(obj.Value) == "string" then obj.Value = { obj.Value } elseif type(obj.Value) ~= "table" then obj.Value = {} end
            end
            local E = element({ Title = o.Title or "Dropdown", Desc = o.Desc, Offset = 150 })
            local b = box(E, 140, 36, true)
            local lbl = New("TextLabel", { Size = UDim2.new(1, -36, 1, 0), Position = UDim2.new(0, 12, 0, 0), BackgroundTransparency = 1, TextSize = 17, TextXAlignment = Enum.TextXAlignment.Left,

            local function nameOf(v) return type(v) == "table" and (v.Title or v.Name) or v end
            local function isSel(name)
                if obj.Multi then
                    for _, v in ipairs(obj.Value) do if nameOf(v) == name then return true end end
                    return false
                end
                return obj.Value ~= nil and nameOf(obj.Value) == name
            end
            local function display()
                if obj.Multi then
                    local l = {}
                    for _, v in ipairs(obj.Values) do local n = nameOf(v); if isSel(n) then l[#l + 1] = tostring(n) end end
                    lbl.Text = #l > 0 and table.concat(l, ", ") or "--"
                else
                    local n = obj.Value ~= nil and nameOf(obj.Value) or nil
                    lbl.Text = (n and n ~= "") and tostring(n) or "--"
                end
            end
            local function fire() display(); task.spawn(safe, obj.Callback, obj.Value) end

            local function build()
                for _, it in ipairs(items) do it.btn:Destroy() end

                for idx, v in ipairs(obj.Values) do
                    local name = nameOf(v)

                            if not isSel(name) then
                                table.insert(obj.Value, v)
                            else
                                if not obj.AllowNone and #obj.Value == 1 then return end
                                for k, x in ipairs(obj.Value) do if nameOf(x) == name then table.remove(obj.Value, k); break end end
                            end
                        else
                            obj.Value = v
                        end
                        for _, o2 in ipairs(items) do
                            local s2 = isSel(o2.name)
                            tw(o2.btn, 0.1, { BackgroundTransparency = s2 and 0.67 or 1 })
                            tw(o2.label, 0.1, { TextTransparency = s2 and 0 or 0.4 })
                        end
                        fire()
                    end)
                end
            end
            local function place()
                if not wrap then return end
                local s = S.Scale
                local ap, as = b.AbsolutePosition, b.AbsoluteSize
                local n = #obj.Values
                local h = math.min(400, math.max(46, n * 41 + 5 + 10))
                local w = 250

                local y = ap.Y + as.Y + 6
                local v = viewport()

                wrap.Size = UDim2.fromOffset(w * s, h * s)
                wrap.Menu.Size = UDim2.fromOffset(w, h)

                    build()
                end
                obj.Opened = true; ddOpen = obj
                wrap.Visible = true

                wrap.Menu.Size = UDim2.new(0, 250, 0, 0)
                local fullH = math.min(400, math.max(46, #obj.Values * 41 + 15))
                tw(wrap.Menu, 0.12, { Size = UDim2.fromOffset(250, fullH) }, Enum.EasingStyle.Quart)
                posConn = b:GetPropertyChangedSignal("AbsolutePosition"):Connect(place)
                task.delay(0.15, function()
                    if not obj.Opened then return end
                    outside = UIS.InputBegan:Connect(function(i)
                        if not isPress(i) then return end
                        local p = i.Position
                        local function inside(g) local a, z = g.AbsolutePosition, g.AbsoluteSize; return p.X >= a.X and p.X <= a.X + z.X and p.Y >= a.Y and p.Y <= a.Y + z.Y end
                        if not inside(b) and not inside(wrap) then obj:Close() end
                    end)
                end)
            end
            function obj:Close()
                if not obj.Opened then return end
                obj.Opened = false
                if ddOpen == obj then ddOpen = nil end
                if posConn then posConn:Disconnect(); posConn = nil end
                if outside then outside:Disconnect(); outside = nil end
                if wrap then wrap.Visible = false end
            end
            function obj:Refresh(vals)
                if vals then obj.Values = vals end
                build(); fire()
                if obj.Opened then place() end
            end
            function obj:Select(v)
                if v then obj.Value = v else obj.Value = obj.Multi and {} or nil end
                obj:Refresh()
            end
            function obj:SetValue(v) obj:Select(v) end
            b.MouseButton1Click:Connect(function() if obj.Opened then obj:Close() else obj:Open() end end)
            fire()
            return obj
        end

        return T
    end
