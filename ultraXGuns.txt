local function isThisHubScreen(gui)
    if not gui or not gui:IsA("ScreenGui") then return false end
    if gui.Name == "RobloxGui" then return false end
    if gui.Name == "MobileButtonsGuiUXG" then return true end
    if gui.Name == "MobileButtonsGui" then
        for _, d in ipairs(gui:GetDescendants()) do
            if d:IsA("TextButton") and d.Text and string.find(d.Text, "DASH", 1, true) then return true end end
        return false end
    for _, d in ipairs(gui:GetDescendants()) do
        if d:IsA("TextLabel") and d.Text == HUB_NAME then return true end end
    return false end

local function disableBadScripts(char)
    if not char then return end
    for _, obj in ipairs(char:GetDescendants()) do
        if obj:IsA("LocalScript") then
            local name = obj.Name:lower()
            if name:find("stun") or name:find("fall") or name:find("ragdoll") then
                if not obj.Disabled then
                    obj.Disabled = true
                    table.insert(_disabledScripts, obj) end end end end end

local function enableBadScripts()
    for _, obj in ipairs(_disabledScripts) do
        if obj and obj.Parent then
            pcall(function() obj.Disabled = false end) end end
    table.clear(_disabledScripts) end
